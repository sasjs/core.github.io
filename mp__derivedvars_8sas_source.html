1/**
2 @file
3 @brief Generates the SAS statements that create derived variables
4 @details Derived (computed) variables are configured in a control table, one
5 row per variable. The macro writes the statements that create them to a
6 fileref, so that callers can compose them into a view:
7
8 %mp_derivedvars(libds=&libds,control=perm.derived_vars,outref=dercode)
9 data work.vw_derived / view=work.vw_derived;
10 set &libds;
11 %include dercode;
12 run;
13
14 Alternatively pass outview= and the macro will build the view itself.
15
16 Building a view is what makes a derived variable filterable. A WHERE
17 statement applies to the data set named on the SET statement, so it cannot
18 reference a variable created in the same data step:
19
20 data out;
21 set work.vw_derived;
22 where <filter>;
23 run;
24
25 Each variable produces two statements:
26
27 attrib REGION length=$21 label='REGION=<expression>' format=$21.;
28 REGION = <expression>;
29
30 Type and length are declared in the control table, so the result is
31 deterministic.
32
33 VAR_DEPS lists every input the expression uses - physical columns
34 and derived variables alike. The order the statements are generated in is
35 the dependency order (variables with no unsatisfied dependency first, ties
36 broken by name), so a derived variable may build on another one without the
37 administrator maintaining a sequence number. A cycle is rejected.
38
39 The declaration is checked against the expression: a derived variable that
40 is referenced must be declared, a declared name that is not referenced is an
41 error, and a declared name must be a column of the target table or a derived
42 variable defined for it. Semicolons, macro triggers and comments are
43 rejected, so a row can only ever be an expression. A derived variable may
44 not reuse the name of a physical column, and may not reference itself: an
45 intra-row calculation would imply a retained value, which depends on the
46 order rows happen to arrive in.
47
48 VAR_LABEL is optional. When it is missing or blank, the label is the
49 variable name, an equals sign and the expression - so a reader can see both
50 what the column is and where it comes from. A configured label replaces
51 that, and is used as given.
52
53 The control table is created by mddl_derivedvars.sas, and must carry LIBREF,
54 DSN, VAR_NM, VAR_DEPS, VAR_TYPE, VAR_LEN, VAR_FMT and VAR_LOGIC, with
55 VAR_LABEL optional. Extra columns are ignored, but a control table that
56 holds several versions of the same VAR_NM - for example a bitemporal
57 TX_FROM / TX_TO pair - must be narrowed with where= (typically to the
58 current versions), because two rows with the same name are rejected as a
59 configuration error.

263/**
264 * The identifiers the expression actually references - quoted strings are
265 * removed first, so a name inside a literal is not mistaken for a reference

267data work.&ds_ref.;
268 set work.&ds_byname.;
269 length l $4000 tok $32;
270 l=prxchange("s/'([^']|'')*'//",-1,logic);
271 l=prxchange('s/"([^"]|"")*"//',-1,l);
272 l=prxchange('s/[^A-Za-z0-9_]/ /',-1,upcase(l));

295create table work.&ds_stale. as

311 select 1 from work.&ds_dep. d

380/**
381 * Dependency order. The edges are between derived variables only - a
382 * physical column is available from the start. Repeatedly take the variables
383 * whose derived inputs have all been placed, alphabetically within a round;
384 * if a round places nothing, what is left is a cycle.
385 */
386proc sql;
387create table work.&ds_edge. as
388 select d.name as child, d.dep_nm as parent
389 from work.&ds_dep. d
390 where d.dep_nm in (select name from work.&ds_byname.);
391quit;
392
393data work.&ds_todo.(keep=name);
394 set work.&ds_byname.;
395run;
396
397data work.&ds_placed.(keep=name seq);
398 length name $32;
399 name='';
400 seq=0;
401 if 0;
402run;
403
404%let left=%mf_nobs(work.&ds_todo.);
405%let nplaced=0;
406%let cyc_found=0;
407
408%do %while (&left>0);
409 proc sql;
410 create table work.&ds_ready. as
411 select t.name
412 from work.&ds_todo. t
413 where not exists (
414 select 1 from work.&ds_edge. e
415 inner join work.&ds_todo. t2 on e.parent=t2.name
416 where e.child=t.name
417 )
418 order by t.name;
419 quit;
420
421 %let nready=%mf_nobs(work.&ds_ready.);
422 %if &nready=0 %then %do;
423 /* walk the edges, so the message names the cycle rather than just the
424 names that could not be placed */
425 data _null_;
426 set work.&ds_todo.(obs=1);
427 call symputx('cyc_start',name);
428 run;
429 %let cyc_path=&cyc_start;
430 %let cyc_cur=&cyc_start;
431 %let cyc_seen=%str(.&cyc_start..);
432 %do cyc_i=1 %to &left;
433 %let cyc_next=;
434 proc sql noprint;
435 select e.parent into: cyc_next
436 from work.&ds_edge. e
437 where e.child="&cyc_cur"
438 and e.parent in (select name from work.&ds_todo.);
439 quit;
440 %if %length(&cyc_next)=0 %then %let cyc_i=&left;
441 %else %do;
442 %let cyc_path=&cyc_path. -> &cyc_next;
443 %if %index(&cyc_seen,%str(.&cyc_next..))>0 %then %let cyc_i=&left;
444 %else %do;
445 %let cyc_seen=&cyc_seen.&cyc_next..;
446 %let cyc_cur=&cyc_next;
447 %end;
448 %end;
449 %end;
450 /* leave the loop before aborting - an abort cancels the step, and the
451 loop would re-enter with the ready count still zero */
452 %let cyc_found=1;
453 %let left=0;
454 %end;
455
456 %if &cyc_found=0 %then %do;
457 /* seq carries the dependency order - the placement rounds are not name
458 ordered, so the final merge must not rely on by name alone */
459 data work.&ds_ready.;
460 set work.&ds_ready.;
461 seq=_n_+&nplaced;
462 run;
463 proc append base=work.&ds_placed. data=work.&ds_ready. force;
464 run;
465 %let nplaced=%eval(&nplaced+&nready);
466
467 proc sql;
468 delete from work.&ds_todo.
469 where name in (select name from work.&ds_ready.);
470 quit;
471
472 %let left=%mf_nobs(work.&ds_todo.);
473 %end;
474%end;

476%mp_abort(iftrue= (&cyc_found=1)
477 ,mac=&sysmacroname
478 ,msg=%str(VAR_DEPS contains a cycle: &cyc_path)

495/**
496 * Generate the attrib + assignment statements
497 */
498data _null_;
499 file &outref;
500 set work.&ds_srt.;
501 /* 512 so the doubled quotes fit before the 256 boundary is applied */
502 length stmt $4600 label $512 fmt2 $49;
503 /* the configured label wins when there is one, otherwise the label is the
504 name, an equals sign and the expression. Labels are limited to 256 chars:
505 the quotes are doubled first, then the cut is made on an even quote
506 boundary - truncating between the two halves of an escape would leave the
507 generated literal unterminated */
508 if not missing(lbl) then label=lbl;
509 else label=cats(name,'=',translate(logic,' ','0D0A'x));
510 label=tranwrd(label,"'","''");
511 if length(label)>256 then do;
512 label=substr(label,1,256);
513 if mod(count(label,"'"),2) ne 0 then label=substr(label,1,255);
514 end;
515 if not missing(fmt) and index(fmt,'.')=0 then fmt=cats(fmt,'.');
516 if type='C' then do;
517 if missing(fmt) then fmt2=cats('$',len,'.');
518 else fmt2=fmt;
519 stmt=catx(' ','attrib',name,cats('length=$',len)
520 ,cats("label='",label,"'"),cats('format=',fmt2,';'));
521 end;
522 else do;
523 if missing(fmt) then fmt2=cats(len,'.');
524 else fmt2=fmt;
525 stmt=catx(' ','attrib',name,cats('length=',len)
526 ,cats("label='",label,"'"),cats('format=',fmt2,';'));
527 end;
528 put stmt;
529 stmt=cats(name,' = ',logic,';');
530 put stmt;
531run;

132/* the statements are generated in dependency order, not in the order the
133 rows were loaded - BAND must be assigned before BAND_AGE */
134data work.stmts;

140data work.chkord;
141 set work.stmts end=eof;
142 retain band_seq bandage_seq;
143 if prxmatch('/^BAND\s*=/',strip(line)) and missing(band_seq) then band_seq=seq;
144 if prxmatch('/^BAND_AGE\s*=/',strip(line)) and missing(bandage_seq)
145 then bandage_seq=seq;
146 if eof;
147 if missing(band_seq) or missing(bandage_seq) or band_seq>=bandage_seq
148 then output;
149run;

321/* the assignment order must be v1, then v2 and v3, then v4, then v5 */
322data work.chkorder;
323 infile derref8 truncover;
324 input line $4000.;
325 seq=_n_;
326 length nm $32;
327 if prxmatch('/^V\d+\s*=/',strip(line)) then nm=scan(strip(line),1,'=');
328 else nm='';
329run;
330
331data work.chkorder2;
332 set work.chkorder end=eof;
333 retain s1 s2 s3 s4 s5;
334 if nm='V1' then s1=seq;
335 if nm='V2' then s2=seq;
336 if nm='V3' then s3=seq;
337 if nm='V4' then s4=seq;
338 if nm='V5' then s5=seq;
339 if eof;
340 if missing(s1) or missing(s2) or missing(s3) or missing(s4) or missing(s5)
341 then output;
342 else if s1 ge min(s2,s3) then output;
343 else if max(s2,s3) ge s4 then output;
344 else if s4 ge s5 then output;
345run;
346
347%mp_assertdsobs(work.chkorder2,
348 desc=The diamond graph is assigned in dependency order,
349 test=EQUALS 0,
350 outds=work.test_results

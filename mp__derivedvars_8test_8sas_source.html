346data work.chkorder;
347 infile derref8 truncover;
348 input line $4000.;
349 seq=_n_;

352 else nm='';
353run;
354
355data work.chkorder2;
356 set work.chkorder end=eof;
357 retain s1 s2 s3 s4 s5;
358 if nm='V1' then s1=seq;
359 if nm='V2' then s2=seq;
360 if nm='V3' then s3=seq;
361 if nm='V4' then s4=seq;
362 if nm='V5' then s5=seq;
363 if eof;
364 if missing(s1) or missing(s2) or missing(s3) or missing(s4) or missing(s5)
365 then output;
366 else if s1 ge min(s2,s3) then output;
367 else if max(s2,s3) ge s4 then output;
368 else if s4 ge s5 then output;
369run;

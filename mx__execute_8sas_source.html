1/**
2 @file
3 @brief Will execute a SASjs web service on SAS 9, Viya or SASjs Server
4 @details Prepares the input files and retrieves the resulting datasets from
5 the response JSON.
6
7 Note - the _webout fileref should NOT be assigned prior to running this macro.
8
9 @param [in] program The _PROGRAM endpoint to test
10 @param [in] inputfiles=(0) A list of space seperated fileref:filename pairs as
11 follows:
12 inputfiles=inref:filename inref2:filename2
13 @param [in] inputdatasets= (0) All datasets in this space seperated list are
14 converted into SASJS-formatted CSVs (see mp_ds2csv.sas) files and added to
15 the list of `inputfiles` for ingestion. The dataset will be sent with the
16 same name (no need for a colon modifier).
17 @param [in] inputparams=(0) A dataset containing name/value pairs in the
18 following format:
19 |name:$32|value:$1000|
20 |---|---|
21 |stpmacname|some value|
22 |mustbevalidname|can be anything, oops, %abort!!|
23
24 @param [in] debug= (131) Provide the _debug value.
25 @param [in] mdebug= (0) Set to 1 to provide macro debugging
26 @param [in] viyaresult= (WEBOUT_JSON) The Viya result type to return. For
27 more info, see mv_getjobresult.sas
28 @param [in] viyacontext= (SAS Job Execution compute context) The Viya compute
29 context on which to run the service
30 @param [out] outlib= (0) Output libref to contain the final tables. Set to
31 0 if the service output is not in JSON format.
32 @param [out] outref= (0) Output fileref to create, to contain the full _webout
33 response.
34 @param [out] outlogds= (_null_) Name of a dataset to receive the child's log
35 (one column, line). Populated on Viya, from the job log mv_jobflow
36 collects, and on SASjs Server, from the log the server appends to the
37 response - ms_testservice discards that log unless this is set. Not
38 populated on SAS 9, where proc stp returns no child log. On SASjs Server
39 it is only populated when debug is 131 or above: the server appends the log
40 to the response in debug mode, so a lower value returns an empty dataset.
41 @param [out] outstatus= (0) Name of a dataset to receive the outcome of each
42 child run - one row per program, with columns STATE, URI, ERRMSG and
43 PLATFORM. ERRMSG is only populated on SASjs Server, from the abort block;
44 the Viya job flow carries a state but no message, and SAS 9 gives neither.
45
46 <h4> SAS Macros </h4>
47 @li mf_getplatform.sas
48 @li mf_getuniquefileref.sas
49 @li mf_getuniquename.sas
50 @li mp_abort.sas

57
58 <h4> Related Programs </h4>
59 @li mx_execute.test.sas
60
61 @version 9.4
62 @author Allan Bowe
63
64**/

66%macro mx_execute(program,
67 inputfiles=0,
68 inputdatasets=0,
69 inputparams=0,
70 debug=131,
71 mdebug=0,
72 outlib=0,
73 outref=0,
74 outlogds=_null_,
75 outstatus=0,
76 viyaresult=WEBOUT_JSON,
77 viyacontext=SAS Job Execution compute context

79%local dbg pcnt fref1 fref2 webref webrefpath i webcount var platform jobds respref clearmeref;
80%if &mdebug=1 %then %do;
81 %put &sysmacroname entry vars:;
82 %put _local_;
83%end;
84%else %let dbg=*;

86/* sanitise inputparams */
87%let pcnt=0;
88%if &inputparams ne 0 %then %do;
89 data _null_;
90 set &inputparams;
91 if not nvalid(name,'v7') then putlog (_all_)(=);
92 else if name in (
93 'program','inputfiles','inputparams','debug','outlib','outref'
94 ) then putlog (_all_)(=);
95 else do;
96 x+1;
97 call symputx(name,quote(cats(value)),'l');
98 call symputx(cats('pval',x),name,'l');
99 call symputx('pcnt',x,'l');
100 end;
101 run;
102 %mp_abort(iftrue= (%mf_nobs(&inputparams) ne &pcnt)
103 ,mac=&sysmacroname
104 ,msg=%str(Invalid values in &inputparams)
105 )
106%end;
107
108/* convert inputdatasets to filerefs */
109%if "&inputdatasets" ne "0" %then %do;
110 %if %quote(&inputfiles)=0 %then %let inputfiles=;
111 %do i=1 %to %sysfunc(countw(&inputdatasets,%str( )));
112 %let var=%scan(&inputdatasets,&i,%str( ));
113 %local dsref&i;
114 %let dsref&i=%mf_getuniquefileref();
115 %mp_ds2csv(&var,outref=&&dsref&i,headerformat=SASJS)
116 %let inputfiles=&inputfiles &&dsref&i:%scan(&var,-1,.);
117 %end;
118%end;

120%let platform=%mf_getplatform();
121%let fref1=%mf_getuniquefileref();
122%let fref2=%mf_getuniquefileref();
123%let webref=%mf_getuniquefileref();
124%let webrefpath=%sysfunc(pathname(work))/%mf_getuniquename();
125/* mp_chop requires a physical path as input */
126filename &webref "&webrefpath";
127
128%if &platform=SASMETA %then %do;
129
130 /* parse the input files */
131 %if %quote(&inputfiles) ne 0 %then %do;
132 %let webcount=%sysfunc(countw(&inputfiles));
133 %put &=webcount;
134 %do i=1 %to &webcount;
135 %let var=%scan(&inputfiles,&i,%str( ));
136 %local webfref&i webname&i;
137 %let webref&i=%scan(&var,1,%str(:));
138 %let webname&i=%scan(&var,2,%str(:));
139 %put webref&i=&&webref&i;
140 %put webname&i=&&webname&i;
141 %end;
142 %end;
143 %else %let webcount=0;
144
145 proc stp program="&program";
146 inputparam _program="&program"
147 %do i=1 %to &webcount;
148 %if &webcount=1 %then %do;
149 _webin_fileref="&&webref&i"
150 _webin_name="&&webname&i"
151 %end;
152 %else %do;
153 _webin_fileref&i="&&webref&i"
154 _webin_name&i="&&webname&i"
155 %end;
156 %end;
157 _webin_file_count="&webcount"
158 _debug="&debug"
159 %do i=1 %to &pcnt;
160 /* resolve name only, proc stp fetches value */
161 &&pval&i=&&&&&&pval&i
162 %end;
163 ;
164 %do i=1 %to &webcount;
165 inputfile &&webref&i;
166 %end;
167 outputfile _webout=&webref;
168 run;
169
170 data _null_;
171 infile &webref;
172 file &fref1;
173 input;
174 length line $10000;
175 if index(_infile_,'>>weboutBEGIN<<') then do;
176 line=tranwrd(_infile_,'>>weboutBEGIN<<','');
177 put line;
178 end;
179 else if index(_infile_,'>>weboutEND<<') then do;
180 line=tranwrd(_infile_,'>>weboutEND<<','');
181 put line;
182 stop;
183 end;
184 else put _infile_;
185 run;
186 data _null_;
187 infile &fref1;
188 input;
189 put _infile_;
190 run;
191 %if &outlib ne 0 %then %do;
192 libname &outlib json (&fref1);
193 %end;
194 %if &outref ne 0 %then %do;
195 filename &outref temp;
196 %mp_binarycopy(inref=&webref,outref=&outref)
197 %end;
198
199 %if "&outlogds" ne "_null_" %then %do;
200 /* SAS 9 returns no child log. Create the dataset all the same, so a caller
201 * that reads it gets no rows rather than a missing table. */
202 data &outlogds;
203 length line $32767;
204 stop;
205 run;
206 %end;
207
208 /* proc stp gives no job status - say so rather than imply success */
209 %if "&outstatus" ne "0" %then %do;
210 data &outstatus;
211 length state $20 uri $256 errmsg $256 platform $10;
212 state='unknown';
213 uri='';
214 errmsg='proc stp returns no job status';
215 platform='SASMETA';
216 output;
217 run;
218 %end;
219
220%end;
221%else %if &platform=SASVIYA %then %do;
222
223 /* prepare inputparams */
224 %local ds1;
225 %let ds1=%mf_getuniquename();
226 %if "&inputparams" ne "0" %then %do;
227 proc transpose data=&inputparams out=&ds1;
228 id name;
229 var value;
230 run;
231 %end;
232 %else %do;
233 data &ds1;run;
234 %end;
235
236 /* parse the input files - convert to sasjs params */
237 %local webcount i var sasjs_tables;
238 %if %quote(&inputfiles) ne 0 %then %do;
239 %let webcount=%sysfunc(countw(&inputfiles));
240 %put &=webcount;
241 %do i=1 %to &webcount;
242 %let var=%scan(&inputfiles,&i,%str( ));
243 %local webfref&i webname&i sasjs&i.data;
244 %let webref&i=%scan(&var,1,%str(:));
245 %let webname&i=%scan(&var,2,%str(:));
246 %put webref&i=&&webref&i;
247 %put webname&i=&&webname&i;
248
249 %let sasjs_tables=&sasjs_tables &&webname&i;
250 data _null_;
251 infile &&webref&i lrecl=32767;
252 input;
253 if _n_=1 then call symputx("sasjs&i.data",_infile_);
254 else call symputx(
255 "sasjs&i.data",cats(symget("sasjs&i.data"),'0D0A'x,_infile_)
256 );
257 putlog "&sysmacroname infile: " _infile_;
258 run;
259 data &ds1;
260 set &ds1;
261 length sasjs&i.data $32767 sasjs_tables $1000;
262 sasjs&i.data=symget("sasjs&i.data");
263 sasjs_tables=symget("sasjs_tables");
264 run;
265 %end;
266 %end;
267 %else %let webcount=0;
268
269 data &ds1;
270 retain _program "&program";
271 retain _contextname "&viyacontext";
272 set &ds1;
273 putlog "&sysmacroname inputparams:";
274 putlog (_all_)(=);
275 run;
276
277 /* the flow table is an internal detail: unique name, thrown away below */
278 %let jobds=%mf_getuniquename(prefix=mx_execute_);
279 %mv_jobflow(inds=&ds1
280 ,maxconcurrency=1
281 ,outds=&jobds
282 ,outref=&fref1
283 ,mdebug=&mdebug
284 )
285 /* show the log */
286 data _null_;
287 infile &fref1;
288 input;
289 putlog _infile_;
290 run;
291 /* and keep it, if the caller asked */
292 %if "&outlogds" ne "_null_" %then %do;
293 data &outlogds;
294 infile &fref1 lrecl=32767 end=eof;
295 length line $32767;
296 do while(not eof);
297 input;
298 output;
299 end;
300 run;
301 %end;
302 /* get the uri to fetch results */
303 data _null_;
304 set &jobds;
305 call symputx('uri',uri);
306 putlog "&sysmacroname: fetching results for " uri;
307 run;
308 /* the outcome, before the flow table goes away */
309 %if "&outstatus" ne "0" %then %do;
310 data &outstatus;
311 length state $20 uri $256 errmsg $256 platform $10;
312 platform='SASVIYA';
313 errmsg='';
314 set &jobds(keep=state uri);
315 run;
316 %end;
317 /* nothing else needs the flow table - do not leave it in the caller's WORK */
318 proc datasets lib=work nolist nowarn;
319 delete &jobds;
320 quit;
321 /* fetch results from webout.json */
322 %mv_getjobresult(uri=&uri,
323 result=&viyaresult,
324 outref=&outref,
325 outlib=&outlib,
326 mdebug=&mdebug
327 )
328
329%end;
330%else %if &platform=SASJS %then %do;
331
332 /* the response is needed to judge the outcome, even if not requested -
333 * and only a ref we create here may be cleared afterwards */
334 %let respref=&outref;
335 %if "&respref"="0" and "&outstatus" ne "0" %then %do;
336 %let respref=%mf_getuniquefileref();
337 %let clearmeref=1;
338 %end;
339 %ms_testservice(&program
340 ,inputfiles=&inputfiles
341 ,inputdatasets=&inputdatasets
342 ,inputparams=&inputparams
343 ,debug=&debug
344 ,mdebug=&mdebug
345 ,outlib=&outlib
346 ,outref=&respref
347 ,outlogds=&outlogds
348 )
349 %if "&outstatus" ne "0" %then %do;
350 data &outstatus;
351 length state $20 uri $256 errmsg $256 platform $10;
352 retain state 'completed' uri '' errmsg '' platform 'SASJS' abortseen 0;
353 infile &respref end=eof lrecl=32767;
354 /* scan the whole response - the abort block is not always on the first
355 * line - but only take a message from inside that block, so a service
356 * with its own MSG field does not look like an abort */
357 do while(not eof);
358 input;
359 if index(_infile_,'"sasjsAbort"')>0 then do;
360 state='canceled';
361 abortseen=1;
362 end;
363 if abortseen=1 and errmsg='' and index(_infile_,'"MSG"')>0 then do;
364 /* prxchange returns its input untouched when nothing matches, which
365 * would put the whole line in errmsg */
366 if prxmatch('/"MSG" *: *"([^"]*)"/',_infile_) then
367 errmsg=prxchange('s/.*"MSG" *: *"([^"]*)".*/$1/',-1,_infile_);
368 end;
369 end;
370 /* one row, whatever the response contained */
371 output;
372 run;
373 %end;
374 %if %length(&clearmeref)>0 %then %do;
375 filename &respref clear;
376 %end;
377
378%end;
379%else %do;
380 %put %str(ERR)OR: Unrecognised platform: &platform;
381%end;
382
383%if &mdebug=0 %then %do;
384 filename &fref1 clear;
385 %if &platform ne SASJS %then %do;
386 filename &fref2 clear;
387 filename &webref clear;
388 %end;
389%end;
390%else %do;
391 %put &sysmacroname exit vars:;
392 %put _local_;
393%end;
394
395%mend mx_execute;
